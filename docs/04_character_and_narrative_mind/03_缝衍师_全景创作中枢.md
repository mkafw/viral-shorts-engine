        // --- 视图切换逻辑 ---
        function switchTab(viewId) {
            // 更新按钮状态
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('active');
            });
            document.getElementById(`btn-${viewId}`).classList.add('active');

            // 隐藏所有视图
            const views = ['dashboard', 'system', 'blueprint', 'validation'];
            views.forEach(v => {
                document.getElementById(`view-${v}`).classList.add('hidden');
                document.getElementById(`view-${v}`).classList.remove('block');
            });

            // 显示选中视图
            const targetView = document.getElementById(`view-${viewId}`);
            targetView.classList.remove('hidden');
            targetView.classList.add('block');

            state.currentView = viewId;

            // 视图特定逻辑
            if (viewId === 'dashboard') {
                renderCharts(); // 重新渲染以适应容器
            } else if (viewId === 'blueprint') {
                renderTimeline();
            }
        }

        // --- 质检表逻辑 ---
        function renderChecklist() {
            const container = document.getElementById('checklist-container');
            container.innerHTML = '';
            
            state.checklist.forEach(item => {
                const checkedClass = item.checked ? 'bg-emerald-900/20 border-emerald-500/30' : 'bg-slate-900 border-slate-700';
                const textClass = item.checked ? 'text-slate-400 line-through' : (item.isCore ? 'text-rose-400 font-bold' : 'text-slate-300');
                const checkIcon = item.checked ? '✅' : '⬜';
                
                const div = document.createElement('div');
                div.className = `p-3 md:p-4 rounded-lg border cursor-pointer transition-all duration-200 flex items-start gap-3 hover:border-slate-500 ${checkedClass}`;
                div.onclick = () => toggleCheck(item.id);
                
                div.innerHTML = `
                    <div class="mt-1">${checkIcon}</div>
                    <div>
                        <span class="text-xs px-2 py-0.5 rounded bg-slate-800 text-slate-400 mb-1 inline-block">${item.category}</span>
                        <p class="text-sm md:text-base ${textClass}">${item.label}</p>
                    </div>
                `;
                container.appendChild(div);
            });
            updateProgress();
        }

        function toggleCheck(id) {
            const item = state.checklist.find(i => i.id === id);
            if(item) {
                item.checked = !item.checked;
                renderChecklist();
            }
        }

        function updateProgress() {
            const total = state.checklist.length;
            const checked = state.checklist.filter(i => i.checked).length;
            const percent = (checked / total) * 100;
            
            document.getElementById('check-progress').innerText = `${checked} / ${total}`;
            document.getElementById('progress-bar').style.width = `${percent}%`;

            const coreItem = state.checklist.find(i => i.isCore);
            const warningEl = document.getElementById('validation-result');
            const successEl = document.getElementById('success-result');

            if (checked === total) {
                warningEl.classList.add('hidden');
                successEl.classList.remove('hidden');
            } else if (checked > total - 2 && !coreItem.checked) {
                warningEl.classList.remove('hidden');
                successEl.classList.add('hidden');
            } else {
                warningEl.classList.add('hidden');
                successEl.classList.add('hidden');
            }
        }

        function resetChecklist() {
            state.checklist.forEach(i => i.checked = false);
            renderChecklist();
        }
